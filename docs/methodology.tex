\section{Données et rendements}\label{sec:data}

\subsection{Rendement simple et rendement logarithmique}

\begin{definition}
Pour un prix $P_t$, le rendement simple et le log-rendement sont
\[
R_t=\frac{P_t}{P_{t-1}}-1,\qquad r_t=\ln\frac{P_t}{P_{t-1}}=\ln(1+R_t).
\]
\end{definition}

Leurs propriétés d'agrégation sont opposées.

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
& Agrégation en coupe (entre actifs) & Agrégation dans le temps \\
\midrule
Simple $R$ & $R_{p,t}=\sum_i w_iR_{i,t}$ \quad exact & $\prod_t(1+R_t)-1$ \\
Log $r$ & $r_{p,t}\neq\sum_i w_ir_{i,t}$ & $\sum_t r_t$ \quad exact \\
\bottomrule
\end{tabular}
\end{center}

\begin{proposition}\label{prop:simple}
Le rendement simple d'un portefeuille est la moyenne pondérée des rendements simples
des actifs, $R_{p,t}=w\T R_t$, avec $w$ les poids en valeur en $t-1$. La propriété
est fausse pour les log-rendements.
\end{proposition}

\begin{proof}
La valeur du portefeuille est $V_t=\sum_i n_iP_{i,t}$ pour des quantités $n_i$ fixes
sur la période. Alors
\[
\frac{V_t}{V_{t-1}}=\sum_i\frac{n_iP_{i,t-1}}{V_{t-1}}\cdot\frac{P_{i,t}}{P_{i,t-1}}
=\sum_i w_i(1+R_{i,t}),
\]
d'où $R_{p,t}=w\T R_t$. Pour les log-rendements, $r_{p,t}=\ln\sum_iw_ie^{r_{i,t}}$,
et par stricte concavité du logarithme (Jensen), $\ln\sum_iw_ie^{r_i}>\sum_iw_ir_i$
dès que les $r_i$ ne sont pas tous égaux.
\end{proof}

\textbf{Conséquence.} Toute construction de portefeuille et tout backtest se font en
rendements simples. Les log-rendements servent à la modélisation statistique (plus
symétriques, additifs dans le temps). Relation utile : $r=\ln(1+R)=R-R^2/2+O(R^3)$,
écart négligeable en quotidien, pas en annuel.

\subsection{Alignement temporel}

Dans \py{portopt}, $R_t$ daté $t$ est le rendement de la clôture $t-1$ à la clôture $t$.
Un poids décidé avec l'information disponible à la clôture $t-1$ gagne $R_t$. C'est cet
alignement qui garantit l'absence de biais d'anticipation (look-ahead) dans le backtest
(section~\ref{sec:backtest}).

\subsection{Nettoyage}

\py{clean_prices} applique trois règles.
\begin{enumerate}
\item \textbf{Forward-fill limité} (\py{ffill_limit=5}) : comble les jours fériés quand
on mélange des places de cotation. Au-delà, un trou est une vraie absence de donnée.
\item \textbf{Jamais de back-fill} : \py{bfill} recopie un prix futur dans le passé,
c'est un biais d'anticipation déguisé.
\item \textbf{Les NaN initiaux sont conservés} : un ETF lancé en 2012 a des NaN avant.
Le backtester ne l'investit qu'une fois qu'il dispose d'une fenêtre in-sample complète.
Supprimer en amont les actifs « incomplets » ne garderait que les survivants :
\emph{biais de survie}.
\end{enumerate}

\subsection{Prix ajustés}

\py{download_prices(auto_adjust=True)} renvoie des clôtures ajustées des splits et des
dividendes, donc des rendements \emph{total return}. Sur des prix bruts, chaque
détachement de dividende apparaît comme une perte : un ETF obligataire perdrait
artificiellement 3 à 4\pct{} par an.

\section{Estimer la covariance}\label{sec:cov}

Toutes les méthodes sauf 1/N consomment $\Sigma$. La qualité de l'allocation est
bornée par la qualité de cet estimateur.

\subsection{Covariance empirique}

\[
S=\frac{1}{T-1}\sum_{t=1}^T(R_t-\bar R)(R_t-\bar R)\T .
\]
$S$ est sans biais mais bruitée dès que $N/T$ n'est pas petit. Elle compte
$N(N+1)/2$ paramètres : 55 pour 10 actifs, 5\,050 pour 100 actifs. Avec $T=252$,
estimer 5\,050 paramètres à partir de 25\,200 nombres n'a pas de sens. Si $T\le N$,
$S$ est de rang au plus $T-1$ : singulière, non inversible.

\begin{theoreme}[\citealp{marchenko1967}]
Si les $R_t$ sont iid de covariance $I_N$ et $N,T\to\infty$ avec $q=N/T\in(0,1]$,
la distribution empirique des valeurs propres de $S$ converge vers une loi à support
\[
\lambda\in\bigl[(1-\sqrt q)^2,\ (1+\sqrt q)^2\bigr].
\]
\end{theoreme}

Avec $q=0.5$, les valeurs propres s'étalent entre $0.09$ et $2.91$ alors qu'elles
valent toutes 1. \textbf{Les grandes valeurs propres sont surestimées, les petites
sous-estimées.} Or toute méthode qui inverse $\Sigma$ (min-variance, Markowitz) donne
le plus de poids aux directions de plus petite variance, précisément celles où
l'estimation est la plus fausse : c'est l'\emph{error maximization} de
\citet{michaud1989}.

\subsection{Ledoit-Wolf}

L'idée est de combiner $S$ (sans biais, forte variance) et une cible structurée
(biaisée, sans variance). Le compromis biais-variance optimal a une forme fermée
\citep{ledoitwolf2004}.

\begin{definition}[estimateur de Ledoit-Wolf]
Soit $S$ la covariance empirique normalisée par $T$\footnote{L'article et le code
utilisent la normalisation $1/T$ pour $S$ dans la formule de shrinkage ; la différence
avec $1/(T-1)$ est d'ordre $1/T$.} et $\lVert A\rVert^2=\tr(AA\T)/N$ la norme de
Frobenius normalisée. On pose $\mu=\tr(S)/N$ et
\[
d^2=\lVert S-\mu I\rVert^2,\qquad
\bar b^2=\frac{1}{T^2}\sum_{t=1}^T\lVert x_tx_t\T-S\rVert^2,\qquad
\delta^\ast=\frac{\min(\bar b^2,d^2)}{d^2},
\]
où $x_t=R_t-\bar R$. L'estimateur est $\hat\Sigma=\delta^\ast\mu I+(1-\delta^\ast)S$.
\end{definition}

\textbf{Interprétation.} $d^2$ mesure l'éloignement de $S$ à la cible (dispersion de ses
valeurs propres). $\bar b^2$ mesure l'incertitude de $S$ (variance d'échantillonnage
estimée à partir des observations). $\delta^\ast$ est la fraction de la dispersion
observée attribuable au bruit : plus $T$ est petit devant $N$, plus on contracte.

\textbf{Propriétés.} Pour $\delta^\ast>0$, $\hat\Sigma$ est définie positive. Les valeurs
propres sont contractées vers $\mu$ \emph{sans changer les vecteurs propres}, et le
conditionnement s'améliore.

\begin{lemme}[calcul rapide de $\bar b^2$]
Avec $S=\frac1T\sum_tx_tx_t\T$,
\[
\sum_{t=1}^T\lVert x_tx_t\T-S\rVert_F^2=\sum_{t=1}^T(x_t\T x_t)^2-T\lVert S\rVert_F^2 .
\]
\end{lemme}

\begin{proof}
$\lVert x_tx_t\T-S\rVert_F^2=(x_t\T x_t)^2-2\,x_t\T Sx_t+\lVert S\rVert_F^2$. Or
$\sum_tx_t\T Sx_t=\tr\bigl(S\sum_tx_tx_t\T\bigr)=T\tr(S^2)=T\lVert S\rVert_F^2$.
En sommant : $\sum_t(x_t\T x_t)^2-2T\lVert S\rVert_F^2+T\lVert S\rVert_F^2$.
\end{proof}

Le terme $\bar b^2$ coûte ainsi $O(TN)$ en plus du calcul de $S$, sans former les $T$
matrices $x_tx_t\T$.

\fig{lw_spectrum.png}{Spectre des valeurs propres, 50 actifs et 75 jours, vrai modèle à
un facteur. La covariance empirique écrase les petites valeurs propres (la dernière est
plus de 10 fois trop petite) ; Ledoit-Wolf les remonte et divise le conditionnement par 5.}

\subsection{EWMA}

\[
\hat\Sigma_t=\sum_{s=0}^{T-1}\omega_s(R_{t-s}-\bar R)(R_{t-s}-\bar R)\T,\qquad
\omega_s\propto\lambda^s,\quad\lambda=2^{-1/h},
\]
avec $h$ la demi-vie (63 jours par défaut), dans l'esprit de RiskMetrics
\citep{riskmetrics1996}. Le nombre effectif d'observations
$(\sum\omega_s)^2/\sum\omega_s^2\approx(1+\lambda)/(1-\lambda)$ vaut environ 180 jours
pour $h=63$ : plus réactif aux changements de régime, mais plus bruité.

\begin{conseil}
\begin{itemize}
\item Le choix de l'estimateur de covariance compte souvent plus que le choix entre ERC
et HRP.
\item La moyenne est bien plus difficile à estimer que la covariance. L'erreur standard
d'une moyenne annualisée est $\sigma/\sqrt{Y}$ avec $Y$ le nombre d'années : avec
$\sigma=16\pct$ et 10 ans, $\pm5\pct$ sur une moyenne de 8\pct. Elle ne dépend pas de
la fréquence d'échantillonnage. L'erreur sur une variance décroît avec le nombre
d'observations : passer en quotidien aide. C'est la raison de fond pour laquelle toutes
les méthodes de ce document ignorent les rendements espérés.
\item Au-delà de 50 actifs, préférer un modèle factoriel $\Sigma=B\Sigma_FB\T+D$ ou le
shrinkage non linéaire \citep{ledoitwolf2020}.
\end{itemize}
\end{conseil}

\py{estimate_cov}. Chaque estimateur accepte

\py{returns -> ndarray}.
\end{code}

\section{La brique commune : contributions au risque}\label{sec:rc}

\subsection{Décomposition d'Euler}

\begin{proposition}\label{prop:euler}
La volatilité $\sigma_p(w)=\sqrt{w\T\Sigma w}$ est homogène de degré 1 et
\[
\sigma_p(w)=\sum_{i=1}^Nw_i\,\frac{\partial\sigma_p}{\partial w_i}(w),\qquad
\frac{\partial\sigma_p}{\partial w}=\frac{\Sigma w}{\sigma_p}.
\]
\end{proposition}

\begin{proof}
$\sigma_p(\lambda w)=\lambda\sigma_p(w)$ pour $\lambda>0$. On dérive par rapport à
$\lambda$ et on évalue en $\lambda=1$ : $\sum_iw_i\,\partial_i\sigma_p(w)=\sigma_p(w)$
(théorème d'Euler pour les fonctions homogènes). Le gradient s'obtient en dérivant
$\sqrt{w\T\Sigma w}$.
\end{proof}

\begin{definition}
Contribution marginale, contribution totale et contribution relative de l'actif $i$ :
\[
\mathrm{MRC}_i=\frac{(\Sigma w)_i}{\sigma_p},\qquad
\mathrm{RC}_i=w_i\frac{(\Sigma w)_i}{\sigma_p},\qquad
\frac{\mathrm{RC}_i}{\sigma_p}=\frac{w_i(\Sigma w)_i}{w\T\Sigma w}.
\]
Par la proposition~\ref{prop:euler}, $\sum_i\mathrm{RC}_i=\sigma_p$ et les
contributions relatives somment à 1.
\end{definition}

\subsection{Interprétation}

Comme $(\Sigma w)_i=\Cov(R_i,R_p)$,
\[
\mathrm{MRC}_i=\frac{\Cov(R_i,R_p)}{\sigma_p}=\rho_{i,p}\,\sigma_i=\beta_{i,p}\,\sigma_p .
\]
Un actif contribue au risque en proportion de son \textbf{bêta au portefeuille}, pas de
sa volatilité propre. Un actif très volatil mais décorrélé du portefeuille contribue
peu ; un actif avec $\rho_{i,p}<0$ a une contribution négative : il réduit le risque.

\begin{exemple}
Portefeuille 60/40 actions/obligations, $\sigma_A=16\pct$, $\sigma_O=6\pct$, $\rho=-0.2$.
\begin{align*}
\sigma_p^2&=0.6^2\cdot0.16^2+0.4^2\cdot0.06^2+2\cdot0.6\cdot0.4\cdot(-0.2)\cdot0.16\cdot0.06\\
&=0.009216+0.000576-0.000922=0.008870,\qquad\sigma_p=9.4\pct,\\
(\Sigma w)_A&=0.6\cdot0.0256+0.4\cdot(-0.00192)=0.014592,\\
\mathrm{RC}_A/\sigma_p&=0.6\cdot0.014592/0.008870=98.7\pct,\qquad
\mathrm{RC}_O/\sigma_p=1.3\pct.
\end{align*}
Un portefeuille 60/40 en capital est un portefeuille 99/1 en risque. C'est la motivation
historique du risk parity.
\end{exemple}

rebalancement.
\end{code}

\subsection{Equal Weight (1/N)}

\[
w_i=\frac1N .
\]

\textbf{Intuition.} Ne rien estimer, donc ne rien estimer de faux.

\textbf{Théorie.} Si les $N$ actifs ont la même volatilité $\sigma$ et la même
corrélation $\rho$,
\[
\sigma_p^2=\frac{\sigma^2}{N}\bigl(1+(N-1)\rho\bigr)\xrightarrow[N\to\infty]{}\rho\sigma^2 :
\]
la diversification élimine le risque idiosyncratique, pas le risque commun.

\textbf{Résultat clé.} \citet{demiguel2009} comparent 14 modèles d'optimisation
(Markowitz, Bayes-Stein, min-variance contrainte, etc.) sur 7 jeux de données :
aucun ne bat significativement 1/N en Sharpe hors échantillon. La fenêtre d'estimation
nécessaire pour que Markowitz batte 1/N est de l'ordre de 3\,000 mois pour 25 actifs.
1/N est le \textbf{benchmark minimal} de toute recherche en allocation.

\textbf{Limites.} Allocation en capital et non en risque : l'actif le plus volatil
domine (section~\ref{sec:rc}). Le portefeuille dépend de la définition de l'univers :
ajouter 5 ETF actions à un univers de 10 le transforme radicalement.

\begin{conseil}
Le turnover de 1/N n'est pas nul : les poids dérivent avec les prix et il faut
rebalancer (38\pct{} par an dans l'exemple de la section~\ref{sec:results}).
Un 1/N en buy-and-hold n'est plus un 1/N au bout de six mois.
\end{conseil}

\subsection{Inverse volatilité}

\[
w_i=\frac{\sigma_i^{-k}}{\sum_j\sigma_j^{-k}},\qquad k=1\ \text{(inverse volatilité)},\quad k=2\ \text{(inverse variance)}.
\]

\textbf{Intuition.} Donner à chaque actif le même budget de volatilité en ignorant les
corrélations : $w_i\sigma_i$ constant. On parle aussi de \emph{naive risk parity}.

\begin{proposition}\label{prop:invvol}
\begin{enumerate}[label=(\roman*)]
\item Si toutes les corrélations sont égales ($\rho_{ij}=\rho$ pour $i\ne j$), le
portefeuille inverse volatilité est le portefeuille ERC.
\item Si $\Sigma$ est diagonale, le portefeuille inverse variance est le portefeuille
de variance minimale.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Avec $w_i=c/\sigma_i$ :
$(\Sigma w)_i=\sum_j\rho_{ij}\sigma_i\sigma_j\frac{c}{\sigma_j}=\sigma_i\,c\,\bigl(1+(N-1)\rho\bigr)$,
donc $w_i(\Sigma w)_i=c^2\bigl(1+(N-1)\rho\bigr)$ ne dépend pas de $i$.
(ii) Le portefeuille de variance minimale est $w\propto\Sigma^{-1}\one$ et, pour
$\Sigma$ diagonale, $\Sigma^{-1}\one=(1/\sigma_i^2)_i$.
\end{proof}

InvVol est donc ERC sous l'hypothèse de corrélations homogènes, et l'inverse variance
est le min-variance sous l'hypothèse de corrélations nulles. L'écart entre InvVol et ERC
mesure \textbf{l'information apportée par la structure de corrélation}.

\begin{conseil}
Avec deux actifs, InvVol et ERC coïncident quelle que soit $\rho$ : il n'y a qu'une
corrélation, elle est donc homogène.
\end{conseil}

\subsection{Risk budgeting}\label{sec:rb}

\textbf{Problème.} Étant donné des budgets $b_i>0$ avec $\sum_ib_i=1$, trouver $w\ge0$,
$\one\T w=1$, tel que
\[
\frac{w_i(\Sigma w)_i}{w\T\Sigma w}=b_i\qquad\forall i .
\]
C'est un système non linéaire ; l'astuce est de le ramener à un problème convexe.

\begin{theoreme}[\citealp{spinu2013,roncalli2013}]\label{thm:spinu}
Soit $\Sigma\succ0$ et
\[
y^\ast=\argmin_{y\in\R^N_{++}}\ f(y),\qquad f(y)=\tfrac12\,y\T\Sigma y-\sum_{i=1}^Nb_i\ln y_i .
\]
Alors $y^\ast$ existe et est unique, et $w^\ast=y^\ast/\one\T y^\ast$ est l'unique
portefeuille long-only de risk budgeting. De plus $y^{\ast\top}\Sigma y^\ast=1$.
\end{theoreme}

\begin{proof}
\emph{Existence.} $f$ est continue sur $\R^N_{++}$. Quand un $y_i\to0^+$, $-b_i\ln y_i\to+\infty$ ;
quand $\lVert y\rVert\to\infty$, $\tfrac12y\T\Sigma y\ge\tfrac12\lambda_{\min}\lVert y\rVert^2$
domine les termes logarithmiques. $f$ est donc coercive et atteint son minimum à
l'intérieur de $\R^N_{++}$.

\emph{Unicité.} $\nabla^2f(y)=\Sigma+\diag(b_i/y_i^2)\succ0$ : $f$ est strictement convexe.

\emph{Condition du premier ordre.} $\partial_if(y)=(\Sigma y)_i-b_i/y_i=0$, soit
$y_i(\Sigma y)_i=b_i$ : les contributions au risque non normalisées de $y^\ast$ sont
exactement les budgets.

\emph{Normalisation.} $y_i(\Sigma y)_i$ est homogène de degré 2, donc les contributions
relatives de $w^\ast=y^\ast/\one\T y^\ast$ valent $b_i/\sum_jb_j=b_i$.

\emph{Unicité du portefeuille.} Si $w$ est un portefeuille de risk budgeting, alors
$y=w/\sqrt{w\T\Sigma w}$ vérifie $y_i(\Sigma y)_i=b_i$, donc annule $\nabla f$, donc
$y=y^\ast$ par stricte convexité.

Enfin, en sommant les conditions du premier ordre, $y^{\ast\top}\Sigma y^\ast=\sum_ib_i=1$.
\end{proof}

\subsubsection*{Algorithme : descente par coordonnées cyclique}

On minimise $f$ en $y_i$ seul, les autres coordonnées fixées
\citep{griveau2013}. La condition du premier ordre en $y_i$ s'écrit
\[
\Sigma_{ii}\,y_i^2+c_i\,y_i-b_i=0,\qquad c_i=\sum_{j\ne i}\Sigma_{ij}\,y_j .
\]
Le discriminant $c_i^2+4\Sigma_{ii}b_i$ est positif et le produit des racines
$-b_i/\Sigma_{ii}$ est négatif : il existe \textbf{une seule racine positive}, d'où la
mise à jour en forme fermée
\[
y_i\leftarrow\frac{-c_i+\sqrt{c_i^2+4\,\Sigma_{ii}\,b_i}}{2\,\Sigma_{ii}} .
\]
La descente par coordonnées avec minimisation exacte converge vers l'optimum global
pour une fonction strictement convexe somme d'un terme lisse et d'un terme séparable
\citep{tseng2001}.

\begin{algorithm}[H]
\caption{Risk budgeting par descente par coordonnées (\py{RiskParity._ccd})}\label{alg:ccd}
\KwIn{$\Sigma\succ0$, budgets $b$, tolérance $\varepsilon=10^{-10}$}
\KwOut{poids $w$}
$y_i\gets b_i/\sigma_i$ pour tout $i$ ; \quad $y\gets y/\sqrt{y\T\Sigma y}$\;
$s\gets\Sigma y$ \Com*[r]{maintenu à jour, jamais recalculé}
\Repeat{$\lVert y-y^{\mathrm{old}}\rVert\le\varepsilon\,\lVert y\rVert$}{
  $y^{\mathrm{old}}\gets y$\;
  \For{$i\gets1$ \KwTo $N$}{
    $c\gets s_i-\Sigma_{ii}y_i$\;
    $y_i'\gets\bigl(-c+\sqrt{c^2+4\Sigma_{ii}b_i}\bigr)/(2\Sigma_{ii})$\;
    $s\gets s+\Sigma_{:,i}\,(y_i'-y_i)$ ; \quad $y_i\gets y_i'$ \Com*[r]{$O(N)$}
  }
}
\KwRet{$w=y/\one\T y$}
\end{algorithm}

Le vecteur $\Sigma y$ est mis à jour incrémentalement : coût $O(N)$ par coordonnée,
$O(N^2)$ par cycle, \textbf{aucune inversion de matrice}. L'initialisation
$y\propto b_i/\sigma_i$ remise à l'échelle $y\T\Sigma y=1$ part près de la solution ;
en pratique 10 à 50 cycles suffisent.

\textbf{Usage des budgets.} Exprimer une vue en risque plutôt qu'en capital. Dans
l'exemple, la stratégie \enquote{RB bond tilt} double le budget de TLT, IEF et LQD :
les trois portent 45\pct{} du risque (15\pct{} chacun) contre 30\pct{} en ERC.

\begin{conseil}
\begin{itemize}
\item Les budgets doivent être strictement positifs (barrière logarithmique). Pour
exclure un actif, retirez-le de l'univers.
\item Sans contrainte long-only, le système $y_i(\Sigma y)_i=b_i$ admet une solution par
orthant, soit jusqu'à $2^{N-1}$ portefeuilles distincts \citep{bai2016}. La solution
long-only est la seule que l'on veuille en pratique.
\item Pour cibler une volatilité (par exemple 10\pct), multipliez $w$ par
$0.10/\sigma_p$ : les contributions relatives sont invariantes au levier.
\end{itemize}
\end{conseil}

\begin{code}
\py{RiskParity(budgets={...})}, méthode \py{_ccd}, attribut \py{n_iter_} (nombre de cycles).
\end{code}

\subsection{Equal Risk Contribution (ERC)}\label{sec:erc}

ERC est le risk budgeting avec $b_i=1/N$ : chaque actif porte la même part du risque.

\begin{theoreme}[\citealp{maillard2010}]\label{thm:erc}
Soit $\sMV$ la volatilité du portefeuille long-only de variance minimale. Alors
\[
\sMV\le\sERC\le\sEW .
\]
\end{theoreme}

\begin{proof}
La première inégalité est immédiate : le min-variance long-only minimise $\sigma_p$ sur
tout le simplexe, qui contient $w_{\mathrm{ERC}}$.

Pour la seconde, $\sigma_p(w)=\lVert\Sigma^{1/2}w\rVert$ est convexe, donc au-dessus de
sa tangente en $w_{\mathrm{ERC}}$ :
\[
\sigma_p\bigl(\tfrac{\one}{N}\bigr)\ \ge\ \sERC+\nabla\sigma_p(w_{\mathrm{ERC}})\T\Bigl(\tfrac{\one}{N}-w_{\mathrm{ERC}}\Bigr)
=\frac1N\sum_i\partial_i\sigma_p(w_{\mathrm{ERC}}),
\]
car $\nabla\sigma_p(w)\T w=\sigma_p(w)$ (proposition~\ref{prop:euler}). En ERC,
$w_i\,\partial_i\sigma_p=\sERC/N$, donc $\partial_i\sigma_p=\sERC/(Nw_i)$ et
\[
\sEW\ \ge\ \frac{\sERC}{N^2}\sum_i\frac1{w_i}\ \ge\ \sERC,
\]
par l'inégalité arithmético-harmonique : $\sum_i1/w_i\ge N^2/\sum_iw_i=N^2$.
\end{proof}

\textbf{Interprétation.} Dans la paramétrisation du théorème~\ref{thm:spinu}, $y^\ast$
minimise aussi $\sqrt{y\T\Sigma y}$ sous la contrainte $\sum_i\ln y_i\ge c$ (la valeur de
$c$ ne change que l'échelle de $y^\ast$). ERC est donc un portefeuille de variance
minimale sous \textbf{contrainte de diversification} logarithmique : sans la contrainte,
on retrouve le min-variance ; la contrainte empêche tout poids de tendre vers zéro.

\fig{erc_two_assets.png}{Deux actifs, $\sigma_1=18\pct$, $\sigma_2=6\pct$, $\rho=-0.2$.
Le min-variance (5.3\pct) met 14\pct{} du risque sur l'actif risqué, 1/N (8.9\pct) en met
95\pct, ERC (5.7\pct) exactement 50\pct. La volatilité d'ERC est à peine au-dessus du
minimum : on achète beaucoup de diversification pour très peu de variance.}

\begin{proposition}\label{prop:erctangent}
Si tous les actifs ont le même ratio de Sharpe et des corrélations homogènes, ERC est
le portefeuille tangent (Sharpe maximal).
\end{proposition}

\begin{proof}
Le portefeuille tangent est $w\propto\Sigma^{-1}\mu$. Avec $\mu_i=s\,\sigma_i$ et
$\Sigma=DCD$, $D=\diag(\sigma_i)$, $C=(1-\rho)I+\rho\one\one\T$ :
$\Sigma^{-1}\mu=s\,D^{-1}C^{-1}\one$. Or $C\one=(1+(N-1)\rho)\one$, donc
$C^{-1}\one=\one/(1+(N-1)\rho)$ et $w\propto D^{-1}\one=(1/\sigma_i)_i$, qui est ERC
par la proposition~\ref{prop:invvol}.
\end{proof}

\textbf{Pourquoi ERC fonctionne hors échantillon.} Il n'utilise pas les rendements
espérés, la quantité la plus mal estimée. Il n'inverse pas $\Sigma$ : les poids
dépendent continûment et de manière stable de $\Sigma$, alors que le min-variance
amplifie le bruit des petites valeurs propres. Et la proposition~\ref{prop:erctangent}
en fait l'optimum sous une hypothèse d'ignorance raisonnable.

\begin{conseil}
\begin{itemize}
\item ERC surpondère les actifs peu volatils (obligations). Sa performance historique
doit beaucoup au marché haussier obligataire de 1981 à 2020. En 2022, actions et
obligations ont baissé ensemble et les fonds risk parity levérés ont perdu de l'ordre
de 20\pct{} ou plus.
\item Un ERC non levéré est peu volatil (8\pct{} dans l'exemple). La comparaison avec
1/N se fait en Sharpe, ou à volatilité égale.
\end{itemize}
\end{conseil}

\subsection{Hierarchical Risk Parity (HRP)}\label{sec:hrp}

\textbf{Intuition} \citep{lopezdeprado2016}. Markowitz traite tous les actifs comme
des substituts potentiels les uns des autres (graphe complet). Or les actifs ont une
structure hiérarchique : SPY et IWM sont proches, TLT et IEF aussi, et ces deux familles
sont éloignées. HRP alloue d'abord entre familles, puis à l'intérieur.

\subsubsection*{Étape 1 : distance}

\[
d_{ij}=\sqrt{\tfrac12(1-\rho_{ij})}\in[0,1].
\]

\begin{proposition}
$d$ est une distance.
\end{proposition}

\begin{proof}
Soit $z_i$ la série de l'actif $i$ centrée et normalisée à norme 1. Alors
$\rho_{ij}=z_i\T z_j$ et $\lVert z_i-z_j\rVert^2=2(1-\rho_{ij})$, donc
$d_{ij}=\tfrac12\lVert z_i-z_j\rVert$. Positivité, symétrie et inégalité triangulaire
sont héritées de la norme euclidienne ; $d_{ij}=0$ si et seulement si $\rho_{ij}=1$.
\end{proof}

\subsubsection*{Étape 2 : clustering hiérarchique}

Agglomération successive des deux clusters les plus proches. La distance entre clusters
dépend du \py{linkage} : \py{single} (minimum des distances, papier original, sensible à
l'effet de chaîne), \py{complete} (maximum, clusters compacts), \py{average} (moyenne,
bon compromis), \py{ward} (minimise l'augmentation de variance intra-cluster).

\subsubsection*{Étape 3 : quasi-diagonalisation}

On réordonne les actifs selon l'ordre des feuilles du dendrogramme. Les actifs corrélés
deviennent voisins et la matrice de covariance réordonnée concentre ses grandes valeurs
près de la diagonale.

\fig{hrp_steps.png}{Étapes de HRP. Dans l'ordre arbitraire (gauche), la structure est
invisible. Après quasi-diagonalisation (droite), deux blocs apparaissent : taux et crédit
(LQD, TLT, IEF) en haut, actifs risqués (VNQ, EEM, EFA, SPY, IWM) en bas, l'or et les
matières premières entre les deux.}

\subsubsection*{Étape 4 : bissection récursive}

On part de $w=\one$. À chaque découpage d'un cluster en deux sous-clusters $L$ et $R$ :
\[
V_c=\tilde w_c\T\Sigma_c\tilde w_c,\quad
\tilde w_c=\frac{\diag(\Sigma_c)^{-1}}{\one\T\diag(\Sigma_c)^{-1}},\qquad
\alpha=1-\frac{V_L}{V_L+V_R},\quad w_L\leftarrow\alpha\,w_L,\quad w_R\leftarrow(1-\alpha)\,w_R .
\]
$V_c$ est la variance du portefeuille inverse variance du cluster $c$ : chaque
sous-cluster reçoit une part inversement proportionnelle à sa variance. C'est une
allocation inverse variance entre deux super-actifs.

\begin{algorithm}[H]
\caption{Hierarchical Risk Parity (\py{HRP})}\label{alg:hrp}
\KwIn{rendements in-sample, \py{linkage}, \py{split}}
\KwOut{poids $w$}
$\Sigma\gets$ covariance estimée ; \quad $\rho\gets$ corrélation\;
$d_{ij}\gets\sqrt{(1-\rho_{ij})/2}$\;
arbre $\gets$ clustering hiérarchique de $d$ (forme condensée)\;
ordre $\gets$ feuilles de l'arbre \Com*[r]{quasi-diagonalisation}
$w\gets\one$ ; \quad pile $\gets$ [ordre]\;
\While{pile non vide}{
  retirer un cluster $c$ de la pile\;
  \lIf{$|c|=1$}{continuer}
  $(L,R)\gets$ \eIf{\py{split="bisection"}}{deux moitiés de la liste $c$}{deux fils du nœud $c$}
  $\alpha\gets1-V_L/(V_L+V_R)$\;
  $w_L\gets\alpha\,w_L$ ; \quad $w_R\gets(1-\alpha)\,w_R$\;
  ajouter $L$ et $R$ à la pile\;
}
\KwRet{$w$}
\end{algorithm}

\begin{proposition}
(i) Aucune inversion de matrice n'est nécessaire : HRP fonctionne même si $\Sigma$ est
singulière ($T<N$). (ii) Si $N=2$ ou si $\Sigma$ est diagonale, HRP est le portefeuille
inverse variance.
\end{proposition}

\begin{proof}
(i) Seules les diagonales des sous-blocs sont inversées. (ii) Pour $\Sigma$ diagonale,
$V_c=1/S_c$ avec $S_c=\sum_{i\in c}\sigma_i^{-2}$, donc $\alpha=V_R/(V_L+V_R)=S_L/(S_L+S_R)$ :
le cluster $L$ reçoit la part inverse variance de ses actifs. Par récurrence sur l'arbre,
chaque actif reçoit $\sigma_i^{-2}/\sum_j\sigma_j^{-2}$. Le cas $N=2$ est un unique split
entre deux singletons, avec $V_c=\sigma_c^2$.
\end{proof}

\textbf{Deux variantes de découpage} (\py{split=}). \py{"bisection"} (papier original)
coupe la \emph{liste ordonnée} en deux moitiés égales, sans respecter l'arbre : un
cluster naturel de 3 actifs dans un univers de 10 peut être coupé en deux.
\py{"dendrogram"} suit les vrais nœuds du dendrogramme ; c'est la variante recommandée.

\begin{conseil}
\begin{itemize}
\item Le code original de López de Prado passe la matrice \emph{carrée} $d$ à
\py{scipy.cluster.hierarchy.linkage}. SciPy interprète alors chaque ligne comme une
observation et calcule $\tilde d_{ij}=\lVert d_{\cdot,i}-d_{\cdot,j}\rVert_2$, une
distance de distances. Beaucoup d'implémentations reproduisent ce comportement sans le
savoir. \py{portopt} passe la forme condensée (\py{squareform}), donc $d$ elle-même.
\item HRP est instable dans le temps : un petit changement de corrélation peut
réorganiser l'arbre et déplacer beaucoup de poids. Dans l'exemple, son turnover
(120\pct{} par an) vaut 2.5 fois celui d'ERC ; à 20~bps de coûts, cela représente
24~bps de performance par an. Essayez \py{linkage="ward"} ou \py{"average"}.
\item HRP concentre le risque dans le cluster le moins volatil (71\pct{} du risque sur
TLT, IEF et LQD dans l'exemple). C'est un portefeuille défensif plus qu'un portefeuille
diversifié en risque.
\end{itemize}
\end{conseil}

\begin{code}
\py{HRP(linkage="single", split="bisection")} ; attributs \py{order_} (ordre
quasi-diagonal) et \py{linkage_matrix_} (à passer à
\py{scipy.cluster.hierarchy.dendrogram}).
\end{code}

\section{Le backtest walk-forward}\label{sec:backtest}

\fig{walk_forward.png}{Backtest walk-forward à fenêtre glissante. Les poids estimés sur
les rendements jusqu'à $t-1$ sont détenus sur $[t,t+H)$, puis la fenêtre avance de $H$.}

\subsection{Mécanique}

\begin{algorithm}[H]
\caption{Backtest walk-forward (\py{Backtester.run})}\label{alg:bt}
\KwIn{rendements $R_{0..T-1}$, estimateur, $L$, $H$, coût $c$}
\KwOut{rendements hors échantillon nets de coûts}
$w^{\mathrm{drift}}\gets0$ \Com*[r]{départ en cash}
$t\gets L$\;
\While{$t<T$}{
  fenêtre $\gets R_{t-L..t-1}$ (\py{rolling}) ou $R_{0..t-1}$ (\py{expanding})\;
  univers $\gets$ actifs sans NaN sur la fenêtre\;
  $w\gets$ copie de l'estimateur ajustée sur la fenêtre\;
  $\tau\gets\sum_i|w_i-w_i^{\mathrm{drift}}|$\;
  détenir $w$ sur $R_{t..t+H-1}$ avec dérive des poids ; richesse $\times(1-c\,\tau)$ en $t$\;
  $w^{\mathrm{drift}}\gets$ poids dérivés en fin de période ; \quad $t\gets t+H$\;
}
\end{algorithm}

Les poids utilisent l'information jusqu'à $t-1$ et gagnent $R_t$ : pas de biais
d'anticipation. Un test unitaire le vérifie avec un estimateur espion qui enregistre la
dernière date qu'il voit. L'estimateur est copié à chaque date : aucun état ne fuit
d'une fenêtre à l'autre.

\subsection{Dérive des poids}

Entre deux rebalancements, un portefeuille réel ne garde pas des poids constants : ils
dérivent avec les prix. Pour $k=1,\dots,H$,
\[
V_k=\sum_iw_i\prod_{s=1}^k(1+R_{i,s}),\qquad
R_{p,k}=\frac{V_k}{V_{k-1}}-1,\qquad
w_i^{\mathrm{drift}}=\frac{w_i\prod_{s=1}^H(1+R_{i,s})}{V_H}.
\]
L'implémentation est vectorisée : \py{np.cumprod(1 + R, axis=0) @ w} donne tout le
chemin $V_1,\dots,V_H$ d'un coup. Avec \py{drift=False}, le portefeuille est supposé
rebalancé chaque jour vers la cible, gratuitement : c'est plus optimiste que la réalité.

\subsection{Coûts de transaction}

Au rebalancement, le turnover one-way est mesuré contre les \textbf{poids dérivés} :
\[
\tau_t=\sum_i\bigl|w_i^{\mathrm{cible}}-w_i^{\mathrm{drift}}\bigr| .
\]
Mesurer contre la cible précédente est une erreur fréquente : si les actions ont monté de
10\pct{} pendant le mois, un portefeuille ERC doit en vendre même si sa cible n'a pas
bougé. Ignorer la dérive sous-estime le turnover.

La richesse est multipliée par $(1-c\,\tau_t)$ au rebalancement, $c$ étant donné en
points de base (\py{transaction_cost_bps}). Le premier rebalancement part du cash
($\tau=1$). Le turnover annualisé, hors construction initiale, est
\[
\mathrm{Turnover}_{\mathrm{ann}}=\frac{\sum_{t>t_0}\tau_t}{\text{nombre d'années}},
\]
et le coût annuel vaut environ $c\times\mathrm{Turnover}_{\mathrm{ann}}$ : 120\pct{} de
turnover à 10~bps coûtent 12~bps par an.

\begin{conseil}
\begin{itemize}
\item \textbf{$(L,H)$ est un hyperparamètre.} Testez une grille et regardez la stabilité du
classement des méthodes, jamais le meilleur couple : sinon vous sur-ajustez le backtest.
\item \textbf{Timing luck.} Décaler la première date de rebalancement de $0$ à $H-1$ jours
donne $H$ trajectoires différentes pour la même stratégie. Leur dispersion mesure la
part de hasard du résultat ; pour une stratégie mensuelle, elle est souvent du même
ordre que les écarts entre méthodes.
\item \py{window="expanding"} réduit le bruit d'estimation mais réagit lentement aux
régimes. Une fenêtre glissante de 252 à 756 jours est le standard.
\end{itemize}
\end{conseil}

\begin{code}
\py{Backtester(returns, estimator, in_sample, out_of_sample, window, transaction_cost_bps, drift)} ;
\py{.run()} renvoie les rendements hors échantillon nets ; attributs \py{weights_},

\section{Indicateurs de performance}\label{sec:perf}

Toutes les métriques prennent une \py{pd.Series} (une stratégie) ou un
\py{pd.DataFrame} (une colonne par stratégie) de rendements simples périodiques.
\py{summary()} produit le tableau complet.

\subsection{Rendement total et CAGR}

\[
\mathrm{TR}=\prod_{t=1}^T(1+R_t)-1,\qquad\CAGR=(1+\mathrm{TR})^{P/T}-1 .
\]
Le CAGR est le taux constant qui donne la même richesse finale : c'est le rendement
effectivement vécu par l'investisseur. La moyenne arithmétique annualisée est
$\mu_{\mathrm{ann}}=P\,\bar R$.

\subsection{Volatility drag}

\begin{proposition}
Le taux de croissance logarithmique $g=\ln(1+\CAGR)$ vérifie, au second ordre,
\[
g\approx\mu_{\mathrm{ann}}-\frac{\sigma_{\mathrm{ann}}^2}{2}.
\]
\end{proposition}

\begin{proof}
$g=\frac PT\sum_t\ln(1+R_t)$. Avec $\ln(1+R)=R-R^2/2+O(R^3)$ et
$\frac1T\sum_tR_t^2=\hat\sigma^2+\bar R^2\approx\hat\sigma^2$ en quotidien,
$g\approx P\bar R-P\hat\sigma^2/2=\mu_{\mathrm{ann}}-\sigma_{\mathrm{ann}}^2/2$.
\end{proof}

\begin{exemple}
Deux stratégies à $\mu_{\mathrm{ann}}=8\pct$ : à $\sigma=10\pct$, $g\approx7.5\pct$ ;
à $\sigma=30\pct$, $g\approx3.5\pct$. Dans l'exemple de la section~\ref{sec:results},
1/N a $\mu_{\mathrm{ann}}=7.36\pct$ et $\sigma_{\mathrm{ann}}=11.42\pct$ : la formule prédit
$g=7.36\pct-0.65\pct=6.71\pct$, le réalisé est $\ln(1.0693)=6.70\pct$.
\end{exemple}

\begin{conseil}
C'est l'argument le plus solide en faveur de la réduction de variance : à moyenne égale,
moins de volatilité donne plus de richesse finale. La version multi-actifs est le
\emph{rebalancing premium} ou \emph{excess growth rate}
\citep{fernholz1982,boothfama1992} : pour un portefeuille à poids constants,
$g_p-\sum_iw_ig_i\approx\frac12\bigl(\sum_iw_i\sigma_i^2-\sigma_p^2\bigr)\ge0$.
\end{conseil}

\subsection{Ratio de Sharpe}

\[
\SR=\sqrt P\,\frac{\overline{R-r_f}}{\hat\sigma},\qquad
r_{f,\text{période}}=(1+r_{f,\text{ann}})^{1/P}-1 .
\]
Rendement excédentaire par unité de risque total. Il est \textbf{invariant au levier} :
levérer deux fois double l'excès de rendement et la volatilité. Un investisseur
moyenne-variance choisit la stratégie de Sharpe maximal puis ajuste le levier.

\textbf{Annualisation.} Le facteur $\sqrt P$ n'est valide que pour des rendements iid.
Pour des rendements AR(1) d'autocorrélation $\rho_1$ et un horizon de $q$ périodes grand
\citep{lo2002},
\[
\SR_q\approx\SR_1\cdot\sqrt q\cdot\sqrt{\frac{1-\rho_1}{1+\rho_1}} .
\]
Des rendements lissés ($\rho_1>0$ : hedge funds illiquides, private equity) ont un
Sharpe annualisé par $\sqrt P$ surestimé.

\subsection{Ratio de Sortino}

\[
\mathrm{Sortino}=\sqrt P\,\frac{\overline{R-\tau}}{DD_\tau},\qquad
DD_\tau=\sqrt{\frac1T\sum_{t=1}^T\min(R_t-\tau,0)^2} .
\]
Seule la volatilité à la baisse est pénalisée. La somme du dénominateur porte sur
\emph{toutes} les observations (les positives comptent pour zéro) : diviser par le
nombre d'observations négatives est une erreur courante qui gonfle le downside
deviation. Pour une distribution symétrique et une moyenne petite devant $\sigma$,
$DD_0\approx\sigma/\sqrt2$, donc $\mathrm{Sortino}\approx\sqrt2\,\SR$ ; un rapport
nettement supérieur à 1.41 signale une asymétrie positive.

\subsection{Ratio de Calmar}

\[
\mathrm{Calmar}=\frac{\CAGR}{|\MaxDD|} .
\]
Nombre d'années de rendement nécessaires pour compenser la pire perte. Il dépend
fortement de la longueur de l'historique (le MaxDD ne peut qu'empirer avec $T$) : ne
comparer que des stratégies sur la même période.

\subsection{Hit ratio}

\[
\mathrm{Hit}=\frac1T\sum_t\mathbb 1\{R_t>0\} .
\]
Proche de 50 à 55\pct{} pour des stratégies d'allocation en quotidien, et peu informatif
seul : un hit ratio de 40\pct{} avec des gains deux fois plus grands que les pertes est
excellent (trend following), un hit ratio de 90\pct{} peut cacher une vente de
volatilité (petits gains fréquents, pertes rares et énormes).

\py{sharpe_ratio}, \py{sortino_ratio}, \py{calmar_ratio}, \py{hit_ratio}, \py{summary}.
\end{code}

\section{Indicateurs de risque}\label{sec:risk}

\subsection{Volatilité annualisée}

$\sigma_{\mathrm{ann}}=\hat\sigma\sqrt P$. Mesure symétrique : une hausse de 5\pct{}
« coûte » autant qu'une baisse de 5\pct. Elle est insuffisante pour des distributions
asymétriques ou à queues épaisses (section~\ref{sec:moments}).

\fig{report_03_rolling_vol.png}{Volatilité glissante sur 252 jours. La volatilité
réalisée se regroupe par régimes (volatility clustering) : les épisodes de stress
simulés doublent la volatilité de toutes les stratégies, mais l'écart entre stratégies
reste stable.}

\subsection{Drawdown}

\begin{definition}
Avec la richesse $W_t=\prod_{s\le t}(1+R_s)$ et $W_0=1$,
\[
DD_t=\frac{W_t}{\max\bigl(1,\max_{s\le t}W_s\bigr)}-1,\qquad\MaxDD=\min_tDD_t .
\]
La durée maximale sous l'eau est la plus longue période consécutive avec $DD_t<0$.
\end{definition}

Le capital initial compte comme un pic : une stratégie qui perd 10\pct{} le premier jour
a un drawdown de $-10\pct$, même si $W$ n'a jamais dépassé $0.9$.

\fig{drawdown_anatomy.png}{Anatomie d'un drawdown : pic, creux, récupération et durée
sous l'eau (portefeuille 1/N, données simulées).}

\textbf{Asymétrie des pertes.} Récupérer une perte $\ell$ exige un gain $\ell/(1-\ell)$ :
$-20\pct$ demande $+25\pct$, $-50\pct$ demande $+100\pct$.

\begin{conseil}
Pour un mouvement brownien de dérive $\mu$ et de volatilité $\sigma$
\citep{magdonismail2004}, $\mathbb E[\MaxDD]=\sqrt{\pi/2}\,\sigma\sqrt T$ si $\mu=0$, et
$\mathbb E[\MaxDD]\sim\frac{\sigma^2}{2\mu}\ln T$ si $\mu>0$. Le MaxDD dépend donc
mécaniquement de la durée du backtest : $-15\pct$ sur 5 ans n'est pas comparable à
$-15\pct$ sur 20 ans.
\end{conseil}

\fig{report_02_drawdowns.png}{Drawdowns des six stratégies de l'exemple ; le point
marque le drawdown maximal.}

\subsection{VaR et CVaR historiques}

\begin{definition}
Avec $q_\alpha(R)$ le quantile empirique d'ordre $\alpha$ (5\pct{} par défaut),
\[
\VaR_\alpha=-q_\alpha(R),\qquad\CVaR_\alpha=-\mathbb E\bigl[R\mid R\le q_\alpha(R)\bigr],
\]
toutes deux exprimées en perte positive.
\end{definition}

La VaR à 95\pct{} dit : « 95\pct{} des jours, on ne perd pas plus que cela », sans rien
dire des 5\pct{} restants. La CVaR (expected shortfall) dit : « sur les 5\pct{} pires
jours, on perd en moyenne cela ».

\fig{var_cvar.png}{Distribution des rendements quotidiens du portefeuille 1/N. La loi
normale de mêmes moyenne et écart-type sous-estime la queue gauche (zone ombrée).}

\begin{proposition}
La VaR n'est pas sous-additive ; la CVaR est une mesure de risque cohérente
\citep{artzner1999} : monotone, invariante par translation, positivement homogène et
sous-additive.
\end{proposition}

\begin{exemple}[Contre-exemple de sous-additivité]
Deux obligations indépendantes, chacune avec une probabilité de défaut de 4\pct{}
(perte 100, sinon 0). Pour chacune, $\VaR_{95\%}=0$ puisque le défaut est au-delà du
quantile à 5\pct. Pour le portefeuille des deux, les pertes valent 0, 100 et 200 avec
probabilités $92.16\pct$, $7.68\pct$ et $0.16\pct$ : $\VaR_{95\%}=100>0+0$. La VaR
pénalise la diversification. Côté CVaR, chaque obligation a
$\CVaR_{95\%}=(4\pct\times100)/5\pct=80$, et le portefeuille
$(0.16\pct\times200+4.84\pct\times100)/5\pct=103.2\le160$ : la sous-additivité est
respectée.
\end{exemple}

Le cadre FRTB (Bâle III) a remplacé la VaR à 99\pct{} par l'expected shortfall à
97.5\pct{} pour le risque de marché.

\begin{conseil}
La VaR paramétrique gaussienne $-(\mu+z_\alpha\sigma)$ sous-estime systématiquement le
risque de queue. Pour rester paramétrique, utilisez l'expansion de Cornish-Fisher, qui
corrige le quantile par la skewness et la kurtosis.
\end{conseil}

\subsection{Skewness et kurtosis}\label{sec:moments}

\[
\gamma_3=\frac{\mathbb E[(R-\mu)^3]}{\sigma^3},\qquad
\gamma_4=\frac{\mathbb E[(R-\mu)^4]}{\sigma^4},\qquad
\text{kurtosis excédentaire}=\gamma_4-3 .
\]
$\gamma_3<0$ signale une queue gauche épaisse (pertes rares mais grandes : portage, vente
d'options, crédit). $\gamma_4-3>0$ (leptokurtique) signale plus de jours calmes et plus
de jours extrêmes qu'une gaussienne ; en quotidien, 5 à 20 est courant pour les indices
actions.

\begin{conseil}
Ces moments d'ordre élevé sont très mal estimés : un seul jour extrême peut faire
basculer la skewness de $-1$ à $+1$. Regardez-les sur des sous-périodes avant d'en tirer
une conclusion.
\end{conseil}

\subsection{Turnover}

Défini à la section~\ref{sec:backtest}. C'est un indicateur de risque
d'implémentation : un turnover élevé expose aux coûts, à l'impact de marché et à une
capacité limitée.

\section{Significativité statistique}\label{sec:stats}

Un backtest est une réalisation d'un processus aléatoire. Avant de conclure qu'une
stratégie en bat une autre, il faut vérifier que l'écart se distingue du bruit.

\subsection{Erreur standard du ratio de Sharpe}

Pour des rendements iid gaussiens \citep{lo2002}, en unités annuelles sur $Y$ années,
\[
\se(\SRh)\approx\sqrt{\frac{1+\tfrac12\SR_{\text{période}}^2}{Y}}\approx\frac1{\sqrt Y}.
\]

\fig{sharpe_ci.png}{Bande de confiance à 95\pct{} du Sharpe estimé en fonction de la
longueur de l'historique.}

Pour qu'un Sharpe de 0.5 soit significativement positif à 95\pct, il faut
$Y\ge(1.96/0.5)^2\approx15$ ans ; pour un Sharpe de 1, environ 4 ans. La plupart des
backtests de 10 ans ne distinguent pas un Sharpe de 0.5 d'un Sharpe nul.

\begin{exemple}[Conséquence pour l'exemple]
Les Sharpe vont de 0.64 (1/N) à 0.88 (HRP) sur 14 ans, avec une erreur standard
d'environ $1/\sqrt{14}=0.27$. L'écart de 0.24 est inférieur à une erreur standard : on ne
peut pas affirmer que HRP bat 1/N. Le test d'écart de Sharpe entre deux stratégies
corrélées \citep{jobsonkorkie1981,memmel2003} a une erreur standard plus faible, mais du
même ordre ici.
\end{exemple}

\subsection{Probabilistic Sharpe Ratio}

\begin{definition}[\citealp{bailey2012}]
\[
\mathrm{PSR}(\SR^\ast)=\Phi\!\left(\frac{(\SRh-\SR^\ast)\sqrt{T-1}}{\sqrt{1-\gamma_3\,\SRh+\frac{\gamma_4-1}{4}\,\SRh^2}}\right),
\]
où $\SRh$ et $\SR^\ast$ sont exprimés \emph{par période} (non annualisés), $\gamma_3$ est
la skewness et $\gamma_4$ la kurtosis non excédentaire. C'est la probabilité que le vrai
Sharpe dépasse le seuil $\SR^\ast$.
\end{definition}

\textbf{Lecture du dénominateur.} Une skewness négative ou une kurtosis élevée augmentent
la variance de $\SRh$, donc baissent le PSR. Une stratégie de vente de volatilité avec
un beau Sharpe mais $\gamma_3=-3$ aura un PSR bien plus faible qu'une stratégie de trend
following de même Sharpe avec $\gamma_3>0$. \py{summary()} affiche PSR(SR~>~0) ;
au-dessus de 0.95, le Sharpe est significativement positif.

\subsection{Deflated Sharpe Ratio}

Si l'on a testé $K$ variantes (paramètres, univers, méthodes) et gardé la meilleure, son
Sharpe est biaisé à la hausse : même si tous les vrais Sharpe sont nuls, le maximum de
$K$ Sharpe estimés est positif. Le DSR \citep{bailey2014} remplace $\SR^\ast$ dans le PSR
par l'espérance du maximum de $K$ Sharpe nuls :
\[
\SR^\ast_K\approx\sqrt{\Var(\SRh_k)}\left[(1-\gamma_E)\,\Phi^{-1}\!\Bigl(1-\frac1K\Bigr)+\gamma_E\,\Phi^{-1}\!\Bigl(1-\frac1{Ke}\Bigr)\right],
\]
avec $\gamma_E\approx0.5772$ la constante d'Euler-Mascheroni et $\Var(\SRh_k)$ la
variance des Sharpe observés sur les $K$ essais. Avec $K=100$, le crochet vaut
$0.4228\times2.326+0.5772\times2.680\approx2.53$ : il faut un Sharpe observé d'environ
$2.5$ erreurs standard pour ne pas relever du pur data mining.

\begin{conseil}
Notez le nombre de variantes testées. C'est une donnée du backtest aussi importante que
le Sharpe lui-même.
\end{conseil}

\begin{code}
\py{probabilistic_sharpe_ratio(r, sr_benchmark=0.0)} ; le seuil \py{sr_benchmark} est
donné annualisé et converti en interne.
\end{code}

\section{Checklist de l'expert}\label{sec:checklist}

\newlist{checklist}{itemize}{1}
\setlist[checklist]{label=$\square$,leftmargin=1.5em,itemsep=1pt}

\begin{minipage}[t]{0.48\linewidth}
\textbf{\sffamily Données}
\begin{checklist}
\item Prix ajustés (total return), jamais de rendements sur prix bruts.
\item Pas de back-fill, pas de suppression des actifs à historique court.
\item Rendements simples pour le portefeuille.
\end{checklist}

\textbf{\sffamily Estimation}
\begin{checklist}
\item Ratio $N/T$ vérifié ; au-delà de 0.1, shrinkage ou modèle factoriel.
\item Rendements espérés jamais utilisés sans prior fort ou shrinkage massif.
\end{checklist}

\textbf{\sffamily Backtest}
\begin{checklist}
\item Poids calculés avec l'information jusqu'à $t-1$ uniquement.
\item Dérive des poids modélisée, turnover mesuré contre les poids dérivés.
\item Coûts réalistes (5 à 20~bps sur ETF liquides, davantage sur small caps et
émergents).
\item Robustesse testée sur une grille $(L,H)$ et sur les décalages de rebalancement.
\item Benchmark 1/N systématiquement inclus.
\end{checklist}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
\textbf{\sffamily Évaluation}
\begin{checklist}
\item Comparaison en Sharpe, ou à volatilité égale, pas en CAGR brut.
\item Erreur standard du Sharpe calculée avant toute conclusion.
\item PSR pour la non-normalité, DSR si plusieurs variantes ont été testées.
\item Drawdowns et CVaR examinés, pas seulement la volatilité.
\item Contributions au risque inspectées : savoir où est le risque.
\item Sous-périodes et régimes analysés : la performance d'ensemble peut venir d'un seul
épisode.
\end{checklist}
\end{minipage}
